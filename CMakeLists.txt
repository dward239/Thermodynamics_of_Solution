cmake_minimum_required(VERSION 3.16)
project(ThermodynamicsOfSolution VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(thermo_core species.cpp pitzer.cpp solver.cpp speciation.cpp)
target_include_directories(thermo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(thermo_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(thermo main.cpp)
target_link_libraries(thermo PRIVATE thermo_core)
enable_testing()
add_executable(thermo_tests tests/basic_tests.cpp)
target_link_libraries(thermo_tests PRIVATE thermo_core)
add_test(NAME numerical_components COMMAND thermo_tests)
add_executable(equilibrium_tests tests/equilibrium_tests.cpp)
target_link_libraries(equilibrium_tests PRIVATE thermo_core)
add_test(NAME ideal_equilibrium COMMAND equilibrium_tests)
set_tests_properties(ideal_equilibrium PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
