cmake_minimum_required(VERSION 3.16)
project(ThermodynamicsOfSolution VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(thermo_core species.cpp pitzer.cpp solver.cpp speciation.cpp)
target_include_directories(thermo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(thermo_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(aqueous_engine src/aqueous_engine.cpp src/aqueous_solid.cpp src/aqueous_sweep.cpp)
target_include_directories(aqueous_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(aqueous_engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(thermo main.cpp)
target_link_libraries(thermo PRIVATE thermo_core)
add_executable(aqueous_cli src/aqueous_cli.cpp)
target_link_libraries(aqueous_cli PRIVATE aqueous_engine)
enable_testing()
add_executable(thermo_tests tests/basic_tests.cpp)
target_link_libraries(thermo_tests PRIVATE thermo_core)
add_test(NAME numerical_components COMMAND thermo_tests)
add_executable(equilibrium_tests tests/equilibrium_tests.cpp)
target_link_libraries(equilibrium_tests PRIVATE thermo_core)
add_test(NAME ideal_equilibrium COMMAND equilibrium_tests)
set_tests_properties(ideal_equilibrium PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(aqueous_tests tests/aqueous_tests.cpp)
target_link_libraries(aqueous_tests PRIVATE aqueous_engine)
add_test(NAME aqueous_mvp_plus_one COMMAND aqueous_tests)
set_tests_properties(aqueous_mvp_plus_one PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})

add_executable(aqueous_sweep src/aqueous_sweep_cli.cpp)
target_link_libraries(aqueous_sweep PRIVATE aqueous_engine)
add_executable(sweep_tests tests/sweep_tests.cpp)
target_link_libraries(sweep_tests PRIVATE aqueous_engine)
add_test(NAME aqueous_sweeps COMMAND sweep_tests)

add_executable(solid_tests tests/solid_tests.cpp)
target_link_libraries(solid_tests PRIVATE aqueous_engine)
add_test(NAME restricted_ideal_precipitation COMMAND solid_tests)
set_tests_properties(restricted_ideal_precipitation PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(reference_tests tests/reference_tests.cpp)
target_link_libraries(reference_tests PRIVATE aqueous_engine)
add_test(NAME analytic_carbonate_reference COMMAND reference_tests)
set_tests_properties(analytic_carbonate_reference PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
